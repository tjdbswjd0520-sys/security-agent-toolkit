$ ls

$ ls

$ cd backup

$ ls
c.log

$ cd ..

$ cp -r backup backup2

$ ls

$ cd backup2

$ ls
c.log

$ cat c.log
2026-10-14 09:12:00 WARN failed login for admin

$ cd ..

$ cp a.log /backup

$ cp a.log ./backup/

$ ls
a.log  backup/  backup2/  memo.txt

$ cd backup

$ ls
a.log  c.log

$ cd ..

$ cp -i a.log ./backup/

$ cat /backup/a.log

$ cat ./backup/a.log
2026-10-14 09:12:00 WARN failed login for admin

$ ls
a.log  backup/  backup2/  memo.txt

$ cat memo.txt

$ rm memo.txt

$ ls
a.log  backup/  backup2/

$ rm -r backup2

$ ls
a.log  backup/
